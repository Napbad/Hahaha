#ifndef HAHAHA_UTILS_LOG_LOG_COLOR_H
#define HAHAHA_UTILS_LOG_LOG_COLOR_H

#include <string_view>
namespace h3::utils {

/**
 * @brief ANSI color codes for console output.
 */
enum class LogColor {
    BLACK,   /**< Black color. */
    RED,     /**< Red color. */
    GREEN,   /**< Green color. */
    YELLOW,  /**< Yellow color. */
    BLUE,    /**< Blue color. */
    MAGENTA, /**< Magenta color. */
    CYAN,    /**< Cyan color. */
    WHITE,   /**< White color. */
    RESET    /**< Reset to default color. */
};

/**
 * @brief Convert LogColor to its ANSI escape code.
 * @param color The color.
 * @return std::string_view The ANSI code.
 */
inline std::string_view toString(LogColor color) {
    switch (color) {
    case LogColor::BLACK:
        return "\033[30m";
    case LogColor::RED:
        return "\033[31m";
    case LogColor::GREEN:
        return "\033[32m";
    case LogColor::YELLOW:
        return "\033[33m";
    case LogColor::BLUE:
        return "\033[34m";
    case LogColor::MAGENTA:
        return "\033[35m";
    case LogColor::CYAN:
        return "\033[36m";
    case LogColor::WHITE:
        return "\033[37m";
    case LogColor::RESET:
        return "\033[0m";
    default:
        return "\033[0m"; // Default to reset if unknown color
    }
}

} // namespace h3::utils

#endif // HAHAHA_UTILS_LOG_LOG_COLOR_H
